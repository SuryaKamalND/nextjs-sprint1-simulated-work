$ curl -i http://localhost:3000/api/users
HTTP/1.1 200 OK
{"success":true,"data":[{"id":"1","name":"Ava","email":"ava@example.com"},{"id":"2","name":"Ben","email":"ben@example.com"}]}

$ curl -i -X POST http://localhost:3000/api/users -H "Content-Type: application/json" -d '{"name":"Eve","email":"eve@example.com"}'

{"success":true,"data":{"id":"3","name":"Eve","email":"eve@example.com"}}

$ curl -i -X POST http://localhost:3000/api/users -H "Content-Type: application/json" -d '{"name":"E","email":"nope"}'

{"success":false,"error":{"code":"VALIDATION_ERROR","message":"Invalid user data","details":{"name":["name must be at least 2 characters"],"email":["email must be a valid email address"]}}}
